atropos.adapters

Dumb source normalizers: OHLCV bars and tidy (ts, field, value).

Adapters DECLARE metadata and serve RAW fields; the resolver (atropos.data_contract.materialize()) does as-of cuts, causal transforms, lag alignment, splits and validation. No lag or alignment logic lives here: an adapter that "helped" by shifting labels would break the very contract the manifest verifies.

Every field is declared through a atropos.data_contract.FieldMeta: the value labelled t is known at t + t_known_offset. An adapter must say which moment of the bar its ts column labels:

  • convention="close": close/high/low/volume/range_pct are known exactly at the label (offset 0); the OPEN was already known one bar earlier, so it carries -bar_delta.
  • convention="open": open is known at the label (offset 0); close/high/low/volume/range_pct only exist once the bar has closed, so they carry +bar_delta.

Derived-field rule (pinned): a derived field must NEVER carry a BETTER (smaller) t_known_offset than any of its inputs; range_pct = (high - low) / close inherits the CLOSE offset exactly, never the open's.

Normalization here (dumb, not careless): ts must be timezone-aware (naive is fatal, non-UTC converted to UTC); rows are sorted by key and duplicate keys are fatal. Serving is single-series (entity_axis="none"); multi-entity materialization is deferred (KNOWN-ISSUES section 14), so a symbol-less OHLCVAdapter declares metadata but every data-serving method raises.

 65def _require_columns(
 66    frame: pd.DataFrame, required: tuple[str, ...], tag: str
 67) -> None:
 68    """Fatal unless the frame carries exactly ``required``."""
 69    missing = [col for col in required if col not in frame.columns]
 70    extra = [str(col) for col in frame.columns if col not in required]
 71    if missing or extra:
 72        raise ValueError(
 73            f"{tag}: columns must be exactly {list(required)};"
 74            f" missing {missing}, unexpected {extra}"
 75        )

 78def _to_utc_labels(column: pd.Series, tag: str) -> pd.DatetimeIndex:
 79    """UTC DatetimeIndex from a timestamp column; naive is fatal."""
 80    labels = pd.DatetimeIndex(column)
 81    if labels.tz is None:
 82        raise ValueError(f"{tag}: ts must be timezone-aware (UTC)")
 83    return labels.tz_convert("UTC")

 86def _offset_map(
 87    convention: str, bar_delta: pd.Timedelta
 88) -> dict[str, pd.Timedelta]:
 89    """Per-field ``t_known_offset`` for one bar convention."""
 90    if convention == "close":
 91        open_offset = -bar_delta
 92        close_offset = pd.Timedelta(0)
 93    else:
 94        open_offset = pd.Timedelta(0)
 95        close_offset = bar_delta
 96    return {
 97        "open": open_offset,
 98        "close": close_offset,
 99        "high": close_offset,
100        "low": close_offset,
101        "volume": close_offset,
102        "range_pct": close_offset,
103    }

106def _validate_ohlcv_args(
107    convention: str,
108    bar_delta: pd.Timedelta,
109    symbol: str | None,
110    fields_to_serve: tuple[str, ...],
111) -> None:
112    """Loud ``__init__`` validation: convention, delta, symbol, fields."""
113    if convention not in _ALLOWED_CONVENTIONS:
114        raise ValueError(
115            f"OHLCVAdapter: convention must be one of"
116            f" {list(_ALLOWED_CONVENTIONS)}, got {convention!r}"
117        )
118    if not isinstance(bar_delta, pd.Timedelta) or (
119        bar_delta <= pd.Timedelta(0)
120    ):
121        raise ValueError(
122            "OHLCVAdapter: bar_delta must be a positive pd.Timedelta,"
123            f" got {bar_delta!r}"
124        )
125    if symbol is not None and not isinstance(symbol, str):
126        raise ValueError(
127            "OHLCVAdapter: symbol must be a str or None, got"
128            f" {type(symbol).__name__}"
129        )
130    if len(set(fields_to_serve)) != len(fields_to_serve):
131        raise ValueError(
132            f"OHLCVAdapter: fields_to_serve repeats entries:"
133            f" {list(fields_to_serve)}"
134        )
135    unknown = [item for item in fields_to_serve if item not in _SERVED_FIELDS]
136    if unknown:
137        raise ValueError(
138            f"OHLCVAdapter: fields_to_serve must be a subset of"
139            f" {list(_SERVED_FIELDS)}, got unknown {unknown}"
140        )
141
142
143def _validate_tidy_fields(
144    fields: dict[str, FieldMeta], present: set[str]
145) -> None:
146    """Bidirectional fields <-> tidy-field coverage check."""
147    if not fields:
148        raise ValueError(
149            "TidyAdapter: fields must declare at least one field"
150        )
151    declared = set(fields)
152    missing_meta = sorted(present - declared)
153    missing_data = sorted(declared - present)
154    if missing_meta or missing_data:
155        raise ValueError(
156            f"TidyAdapter: field mismatch -- no FieldMeta for"
157            f" {missing_meta}, no tidy data for {missing_data}"
158        )
159    misnamed = sorted(key for key, meta in fields.items() if meta.name != key)
160    if misnamed:
161        raise ValueError(
162            "TidyAdapter: FieldMeta.name must match its mapping key;"
163            f" mismatched keys: {misnamed}"
164        )

167def _as_of_value(
168    series: pd.Series, offset: pd.Timedelta, ts: pd.Timestamp
169) -> float:
170    """Last value whose label ``+ offset`` lands at/below ``ts``."""
171    cutoff = ts - offset
172    position = int(series.index.searchsorted(cutoff, side="right"))
173    position -= 1
174    if position < 0:
175        return float("nan")
176    return float(series.iloc[position])

180# OHLCV bars adapter
181# ---------------------------------------------------------------------------
182class OHLCVAdapter:
183    """Adapter over one OHLCV bars frame (single symbol, flat fields).
184
185    One :class:`atropos.data_contract.FieldMeta` per served field
186    (dtype float, kind ``"bar"``, ``entity_axis="none"``,
187    ``adjustment="none"``) with the convention-dependent
188    ``t_known_offset`` (module conventions table).  ``range_pct =
189    (high - low) / close`` is computed ONCE here (NaN where
190    ``close == 0``, never an exception) and never claims a better
191    offset than its inputs.  With ``symbol=None`` metadata still
192    declares, but every data-serving method raises the deferred
193    multi-entity ``ValueError``.
194
195    Args:
196        bars: frame with exactly ``["ts", "symbol", "open", "high",
197            "low", "close", "volume"]``; tz-aware ``ts`` (naive is
198            fatal, non-UTC converted); sorted here by ``(symbol,
199            ts)``; duplicate ``(ts, symbol)`` rows are fatal.
200        convention: which moment ``ts`` labels, ``"close"`` or
201            ``"open"`` -- see the module conventions table.
202        bar_delta: bar length; a positive ``pd.Timedelta``.
203        symbol: the ONE symbol served as flat series; ``None``
204            defers serving (data methods raise).
205        name: adapter identity recorded by the resolver.
206        version: adapter version recorded by the resolver.
207        calendar: trading-calendar label (informational).
208        fields_to_serve: subset of ``{"open", "high", "low",
209            "close", "volume", "range_pct"}`` (no repeats).
210
211    Raises:
212        ValueError: on malformed ``bars`` (wrong columns, naive
213            ``ts``, duplicates, emptiness, non-numeric values), a
214            ``symbol`` absent from ``bars``, a bad ``convention`` /
215            ``bar_delta``, or a repeated/unknown
216            ``fields_to_serve`` entry.
217
218    Example::
219
220        >>> import pandas as pd
221        >>> ts = pd.date_range(
222        ...     "2024-01-02 14:00", periods=3, freq="1h", tz="UTC"
223        ... )
224        >>> bars = pd.DataFrame(
225        ...     {"ts": ts, "symbol": "SPY", "open": [99.5, 100.5, 101.5],
226        ...      "high": [101.0, 102.0, 103.0], "low": [100.0, 101.0, 102.0],
227        ...      "close": [100.0, 101.0, 102.0],
228        ...      "volume": [1000.0, 1100.0, 900.0]}
229        ... )
230        >>> adapter = OHLCVAdapter(
231        ...     bars, "close", pd.Timedelta("1h"), symbol="SPY",
232        ...     fields_to_serve=("open", "close", "range_pct"),
233        ... )
234        >>> adapter.fields()["open"].t_known_offset == -pd.Timedelta("1h")
235        True
236        >>> adapter.fields()["range_pct"].t_known_offset == pd.Timedelta(0)
237        True
238        >>> adapter.fetch("close", ts[0], ts[2]).tolist()
239        [100.0, 101.0, 102.0]
240        >>> adapter.as_of("open", ts[0])  # open known a bar earlier
241        100.5
242        >>> adapter.as_of("close", ts[0] - pd.Timedelta("1h"))
243        nan
244        >>> adapter.validate_self()
245        []
246    """
247
248    def __init__(
249        self,
250        bars: pd.DataFrame,
251        convention: Literal["open", "close"],
252        bar_delta: pd.Timedelta,
253        *,
254        symbol: str | None,
255        name: str = "ohlcv",
256        version: str = "1.0",
257        calendar: str = "24/7",
258        fields_to_serve: tuple[str, ...] = ("close",),
259    ) -> None:
260        """Validate, normalize, compute range_pct and declare fields."""
261        _validate_ohlcv_args(convention, bar_delta, symbol, fields_to_serve)
262        if not isinstance(bars, pd.DataFrame):
263            raise ValueError(
264                "OHLCVAdapter: bars must be a pd.DataFrame, got"
265                f" {type(bars).__name__}"
266            )
267        _require_columns(bars, _OHLCV_COLUMNS, "OHLCVAdapter bars")
268        if bars.empty:
269            raise ValueError(
270                "OHLCVAdapter: bars must contain at least one row"
271            )
272        frame = bars.loc[:, list(_OHLCV_COLUMNS)].copy()
273        frame["ts"] = _to_utc_labels(frame["ts"], "OHLCVAdapter bars")
274        frame = frame.sort_values(["symbol", "ts"], kind="stable")
275        duplicates = int(frame.duplicated(subset=["ts", "symbol"]).sum())
276        if duplicates:
277            raise ValueError(
278                f"OHLCVAdapter: {duplicates} duplicate (ts, symbol) row(s)"
279            )
280        symbol_set = set(frame["symbol"])
281        if symbol is not None and symbol not in symbol_set:
282            raise ValueError(
283                f"OHLCVAdapter: symbol {symbol!r} is not in bars;"
284                f" available: {sorted({str(item) for item in symbol_set})}"
285            )
286        self.name = name
287        self.version = version
288        self.calendar = calendar
289        self._symbol: str | None = symbol
290        self._served: tuple[str, ...] = fields_to_serve
291        offsets = _offset_map(convention, bar_delta)
292        self._fields = {
293            field_name: FieldMeta(
294                field_name, t_known_offset=offsets[field_name]
295            )
296            for field_name in self._served
297        }
298        closes = frame["close"].to_numpy(dtype=float)
299        frame["range_pct"] = (
300            frame["high"].to_numpy(dtype=float)
301            - frame["low"].to_numpy(dtype=float)
302        ) / np.where(closes != 0.0, closes, np.nan)
303        self._rows = (
304            frame if symbol is None else frame[frame["symbol"] == symbol]
305        )
306        self._labels = pd.DatetimeIndex(self._rows["ts"])
307        self._series: dict[str, pd.Series] = {}
308        if symbol is None:
309            return
310        self._series = {
311            field_name: pd.Series(
312                self._rows[field_name].to_numpy(dtype=float),
313                index=self._labels,
314                name=field_name,
315            )
316            for field_name in self._served
317        }
318
319    def _require_symbol(self) -> str:
320        """Serving guard: multi-entity is deferred while ``None``."""
321        if self._symbol is None:
322            raise ValueError(
323                "OHLCVAdapter: multi-entity serving is"
324                f" {_DEFERRED_NOTE}; construct with symbol=<name>"
325            )
326        return self._symbol
327
328    def _lookup(self, name: str) -> pd.Series:
329        """The served Series for ``name`` (unknown field is fatal)."""
330        self._require_symbol()
331        if name not in self._series:
332            raise ValueError(
333                f"OHLCVAdapter: unknown field {name!r}; serves"
334                f" {list(self._served)}"
335            )
336        return self._series[name]
337
338    def fields(self) -> dict[str, FieldMeta]:
339        """Declared metadata for every served field."""
340        return dict(self._fields)
341
342    def extent(self) -> tuple[pd.Timestamp, pd.Timestamp]:
343        """Inclusive ``(min ts, max ts)`` of the served symbol."""
344        self._require_symbol()
345        first: pd.Timestamp = self._labels.min()
346        last: pd.Timestamp = self._labels.max()
347        return first, last
348
349    def fetch(
350        self, name: str, start: pd.Timestamp, end: pd.Timestamp
351    ) -> pd.Series:
352        """Raw float values of ``name`` with labels in ``[start, end]``."""
353        series = self._lookup(name)
354        return series.loc[start:end]
355
356    def as_of(self, name: str, ts: pd.Timestamp) -> float:
357        """Value of ``name`` known at ``ts`` (NaN when none is).
358
359        Cutoff ``= ts - t_known_offset``, last label ``<=`` cutoff.
360        """
361        series = self._lookup(name)
362        offset = self._fields[name].t_known_offset
363        return _as_of_value(series, offset, ts)
364
365    def validate_self(self) -> list[str]:
366        """Adapter-internal consistency issues ([] when clean).
367
368        NaN closes, nonpositive volume, bars-per-day irregularity.
369        """
370        issues: list[str] = []
371        rows = self._rows
372        nan_closes = int(rows["close"].isna().sum())
373        if nan_closes:
374            issues.append(f"{nan_closes} close value(s) are NaN")
375        nonpositive = int((rows["volume"] <= 0).sum())
376        if nonpositive:
377            issues.append(f"{nonpositive} volume value(s) are nonpositive")
378        per_day = rows.groupby(rows["ts"].dt.floor("D"))["ts"].count()
379        distinct = sorted({int(count) for count in per_day})
380        if len(distinct) > 1:
381            issues.append(
382                f"bars per day varies across distinct counts: {distinct}"
383            )
384        return issues

390class TidyAdapter:
391    """Adapter over a long-format ``(ts, field, value)`` frame.
392
393    The MVP tidy contract has NO entity column: callers pre-collapse
394    entity levels to one series per field.  Every ``field`` value in
395    the frame must have a declared
396    :class:`atropos.data_contract.FieldMeta` here and vice versa
397    (mismatch is fatal, naming the diff).  The frame is normalized
398    to one float Series per field (sorted by ``ts``) in ``__init__``;
399    ``as_of`` uses each field's own ``t_known_offset`` and
400    ``validate_self`` reports non-finite values per field as
401    non-fatal warnings.
402
403    Args:
404        tidy: frame with exactly ``["ts", "field", "value"]``;
405            tz-aware ``ts`` (naive is fatal, non-UTC converted);
406            sorted here by ``(field, ts)``; duplicate ``(ts, field)``
407            pairs are fatal; ``value`` must be float-coercible and
408            every ``field`` value must be a ``str``.
409        fields: one FieldMeta per tidy field (and no more), keyed by
410            the ``field`` value it describes.
411        name: adapter identity recorded by the resolver.
412        version: adapter version recorded by the resolver.
413        calendar: trading-calendar label (informational).
414
415    Raises:
416        ValueError: on malformed ``tidy`` (wrong columns, naive
417            ``ts``, duplicate ``(ts, field)`` pairs, non-str field
418            values, a non-numeric ``value`` column), a non-dict or
419            empty ``fields``, fields/data set mismatches, or a
420            FieldMeta whose ``name`` differs from its key.
421
422    Example::
423
424        >>> import pandas as pd
425        >>> from atropos.data_contract import FieldMeta
426        >>> ts = pd.date_range(
427        ...     "2024-01-02 14:00", periods=3, freq="1h", tz="UTC"
428        ... )
429        >>> tidy = pd.DataFrame(
430        ...     {
431        ...         "ts": list(ts) + list(ts[:2]),
432        ...         "field": ["spread"] * 3 + ["imbalance"] * 2,
433        ...         "value": [0.01, 0.02, 0.03, 1.5, -2.5],
434        ...     }
435        ... )
436        >>> fields = {
437        ...     "spread": FieldMeta("spread"),
438        ...     "imbalance": FieldMeta(
439        ...         "imbalance", t_known_offset=pd.Timedelta("30min")
440        ...     ),
441        ... }
442        >>> adapter = TidyAdapter(tidy, fields=fields)
443        >>> adapter.extent() == (ts[0], ts[2])
444        True
445        >>> adapter.as_of("imbalance", ts[1] + pd.Timedelta("30min"))
446        -2.5
447        >>> adapter.validate_self()
448        []
449    """
450
451    def __init__(
452        self,
453        tidy: pd.DataFrame,
454        *,
455        fields: dict[str, FieldMeta],
456        name: str = "tidy",
457        version: str = "1.0",
458        calendar: str = "24/7",
459    ) -> None:
460        """Validate, normalize and pivot to one Series per field."""
461        if not isinstance(tidy, pd.DataFrame):
462            raise ValueError(
463                "TidyAdapter: tidy must be a pd.DataFrame, got"
464                f" {type(tidy).__name__}"
465            )
466        if not isinstance(fields, dict):
467            raise ValueError(
468                "TidyAdapter: fields must be a dict[str, FieldMeta],"
469                f" got {type(fields).__name__}"
470            )
471        _require_columns(tidy, _TIDY_COLUMNS, "TidyAdapter tidy")
472        frame = tidy.loc[:, list(_TIDY_COLUMNS)].copy()
473        frame["ts"] = _to_utc_labels(frame["ts"], "TidyAdapter tidy")
474        frame = frame.sort_values(["field", "ts"], kind="stable")
475        duplicates = int(frame.duplicated(subset=["ts", "field"]).sum())
476        if duplicates:
477            raise ValueError(
478                f"TidyAdapter: {duplicates} duplicate (ts, field) pair(s)"
479            )
480        present = set(frame["field"])
481        non_str = sorted(
482            {str(item) for item in present if not isinstance(item, str)}
483        )
484        if non_str:
485            raise ValueError(
486                f"TidyAdapter: field values must be str, got {non_str}"
487            )
488        _validate_tidy_fields(fields, present)
489        self.name = name
490        self.version = version
491        self.calendar = calendar
492        self._fields = dict(fields)
493        values = frame["value"].to_numpy(dtype=float)
494        self._series: dict[str, pd.Series] = {}
495        for field_name in sorted(present):
496            mask = (frame["field"] == field_name).to_numpy()
497            labels = pd.DatetimeIndex(frame.loc[mask, "ts"])
498            self._series[field_name] = pd.Series(
499                values[mask], index=labels, name=field_name
500            )
501
502    def _lookup(self, name: str) -> pd.Series:
503        """The served Series for ``name`` (unknown field is fatal)."""
504        if name not in self._series:
505            raise ValueError(
506                f"TidyAdapter: unknown field {name!r}; serves"
507                f" {sorted(self._series)}"
508            )
509        return self._series[name]
510
511    def fields(self) -> dict[str, FieldMeta]:
512        """Declared metadata (verbatim copy of ``fields``)."""
513        return dict(self._fields)
514
515    def extent(self) -> tuple[pd.Timestamp, pd.Timestamp]:
516        """Union ``(min ts, max ts)`` over every served field."""
517        first: pd.Timestamp = min(
518            series.index.min() for series in self._series.values()
519        )
520        last: pd.Timestamp = max(
521            series.index.max() for series in self._series.values()
522        )
523        return first, last
524
525    def fetch(
526        self, name: str, start: pd.Timestamp, end: pd.Timestamp
527    ) -> pd.Series:
528        """Raw float values of ``name`` with labels in ``[start, end]``."""
529        series = self._lookup(name)
530        return series.loc[start:end]
531
532    def as_of(self, name: str, ts: pd.Timestamp) -> float:
533        """Value of ``name`` known at ``ts`` (NaN when none is).
534
535        Uses ``name``'s OWN ``t_known_offset`` as the cutoff shift.
536        """
537        series = self._lookup(name)
538        offset = self._fields[name].t_known_offset
539        return _as_of_value(series, offset, ts)
540
541    def validate_self(self) -> list[str]:
542        """Adapter-internal consistency issues ([] when clean).
543
544        Per-field non-finite (NaN/inf) value counts.
545        """
546        issues: list[str] = []
547        for field_name, series in self._series.items():
548            values = series.to_numpy(dtype=float)
549            non_finite = int(np.count_nonzero(~np.isfinite(values)))
550            if non_finite:
551                issues.append(
552                    f"field {field_name!r}: {non_finite} non-finite value(s)"
553                )
554        return issues
