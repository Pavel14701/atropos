atropos.predictive_model_eval

Predictive-model evaluation: leakage-safe CV, walk-forward OOS, stability.

Evaluation framework for any model that maps features -> next-period return forecasts. All functions take features (feature matrix) and target (outcome) with rows in CHRONOLOGICAL order.

IMPORTANT — positional alignment: every function pairs features with target (or predictions with outcomes) POSITIONALLY. Pandas inputs are converted via .to_numpy() and their indexes are IGNORED — there is no reindexing and no sorting inside. Rows must already be pairwise aligned AND in chronological order before the call.

Functions

purged_kfold_cv contiguous K-fold with purge gaps on both sides of each test block (avoids leakage from overlapping label horizons); optional model scoring. walkforward_model_eval rolling train/test OOS: out-of-sample IC, directional hit rate, sign-PnL t-stat and a PASS / DEGRADED / FAIL verdict. feature_importance_stability are feature importances stable across bootstrap refits? sign_consistency directional accuracy overall and by regime (time thirds + outcome-magnitude terciles). deflated_performance Sharpe deflated for n_strategies trials (Bailey & Lopez de Prado) + drawdown context.

Model contract: model_fn(features_train, target_train) returns a PREDICTOR - any callable mapping features_test -> (n_test,) forecasts. The tooling optionally reads predictor.feature_importances_ (per-feature importances); the bundled ols_model_fn() (linear), knn_model_fn() (distance-weighted k-NN) and kernel_model_fn() (Nadaraya-Watson gaussian kernel) satisfy the full contract with zero dependencies beyond numpy/pandas/scipy.

Run python -m atropos.predictive_model_eval for a synthetic demo.

  71def _as_arrays(
  72    features: Any, target: Any
  73) -> tuple[np.ndarray, np.ndarray, list[str]]:
  74    """Convert features/target to float arrays, dropping NaN rows.
  75
  76    IMPORTANT: alignment is POSITIONAL, never index-based.  Pandas inputs
  77    are converted with ``.to_numpy()`` and their indexes are IGNORED (no
  78    reindexing, no sorting).  Callers must pass rows already pairwise
  79    aligned and in CHRONOLOGICAL order — the purged/walk-forward splits
  80    below are time blocks and silently assume it.
  81    """
  82    if isinstance(features, pd.DataFrame):
  83        names = [str(column) for column in features.columns]
  84        features_arr = features.to_numpy(dtype=float)
  85    else:
  86        features_arr = np.asarray(features, dtype=float)
  87        if features_arr.ndim == 1:
  88            features_arr = features_arr.reshape(-1, 1)
  89        names = [f"f{idx}" for idx in range(features_arr.shape[1])]
  90    target_arr = (
  91        target.to_numpy(dtype=float).reshape(-1)
  92        if isinstance(target, pd.Series)
  93        else np.asarray(target, dtype=float).reshape(-1)
  94    )
  95    if features_arr.shape[0] != target_arr.size:
  96        raise ValueError(
  97            f"features has {features_arr.shape[0]} rows but target has "
  98            f"{target_arr.size} values"
  99        )
 100    ok = np.isfinite(target_arr) & np.isfinite(features_arr).all(axis=1)
 101    return features_arr[ok], target_arr[ok], names

 104def _pearson(left: np.ndarray, right: np.ndarray) -> float:
 105    if left.size < 3 or np.std(left) <= 0 or np.std(right) <= 0:
 106        return float("nan")
 107    return float(np.corrcoef(left, right)[0, 1])

 135def _standardize_fit(
 136    features_arr: np.ndarray,
 137) -> tuple[np.ndarray, np.ndarray, np.ndarray]:
 138    """Standardize columns with train statistics (zero std -> 1.0).
 139
 140    Args:
 141        features_arr: ``(n_rows, n_features)`` raw feature matrix.
 142
 143    Returns:
 144        Tuple ``(standardized, center, spread)``: the standardized
 145        matrix, the per-column mean, and the per-column std with zeros
 146        replaced by 1.0 so constant columns stay finite.
 147    """
 148    center = features_arr.mean(axis=0)
 149    spread = features_arr.std(axis=0)
 150    spread = np.where(spread > 0.0, spread, 1.0)
 151    return (features_arr - center) / spread, center, spread

 154def _pairwise_dist_sq(
 155    test_std: np.ndarray, train_std: np.ndarray
 156) -> np.ndarray:
 157    """Squared euclidean distances between standardized row sets.
 158
 159    Args:
 160        test_std: ``(n_test, n_features)`` standardized query rows.
 161        train_std: ``(n_train, n_features)`` standardized rows.
 162
 163    Returns:
 164        ``(n_test, n_train)`` matrix of squared distances, clipped at
 165        0 so the gram-matrix expansion never yields tiny negatives.
 166    """
 167    test_norms = np.sum(test_std**2, axis=1)[:, None]
 168    train_norms = np.sum(train_std**2, axis=1)[None, :]
 169    cross = test_std @ train_std.T
 170    squared = np.maximum(test_norms + train_norms - 2.0 * cross, 0.0)
 171    return np.asarray(squared, dtype=float)

 174def _r_squared(actual: np.ndarray, predicted: np.ndarray) -> float:
 175    """Coefficient of determination of ``predicted`` against ``actual``.
 176
 177    Args:
 178        actual: ``(n,)`` observed values.
 179        predicted: ``(n,)`` forecasted values.
 180
 181    Returns:
 182        ``1 - SS_res / SS_tot`` (may be negative); 0.0 when ``actual``
 183        is constant, because the ratio is undefined there.
 184    """
 185    total_ss = float(np.sum((actual - float(actual.mean())) ** 2))
 186    if total_ss <= 1e-12:
 187        return 0.0
 188    residual_ss = float(np.sum((actual - predicted) ** 2))
 189    return 1.0 - residual_ss / total_ss

 192def _permutation_importance(
 193    predict_fn: Callable[[Any], np.ndarray],
 194    features_train: np.ndarray,
 195    target_train: np.ndarray,
 196) -> np.ndarray:
 197    """Capped deterministic permutation importance for a predictor.
 198
 199    The evaluation subsample is the full train set, or 256 rows drawn
 200    without replacement (fixed seed 0) when the train set is larger.
 201    Importance of feature ``j`` is the drop in R^2 after shuffling
 202    column ``j`` (fresh generator seeded ``0 + j``), clipped at 0 and
 203    normalized to sum 1; all-zero drops yield the uniform vector.  The
 204    scheme touches no global RNG state, so fits are reproducible.
 205
 206    CAVEAT - the clip ``max(baseline - permuted, 0)``: a negative
 207    baseline R^2 (model worse than the mean predictor) clips to zero
 208    and can hide features whose permutation actually IMPROVES the
 209    model; interpret importances as RELATIVE rankings within one fit,
 210    not as absolute evidence that a feature helps.
 211
 212    Args:
 213        predict_fn: fitted predictor closure (``features -> forecasts``).
 214        features_train: ``(n_train, n_features)`` raw training matrix.
 215        target_train: ``(n_train,)`` training outcomes.
 216
 217    Returns:
 218        ``(n_features,)`` importance vector summing to 1.
 219    """
 220    n_rows, n_features = features_train.shape
 221    if n_rows > _IMPORTANCE_CAP:
 222        subsample = np.random.default_rng(_IMPORTANCE_SEED).choice(
 223            n_rows, size=_IMPORTANCE_CAP, replace=False
 224        )
 225        features_sub = features_train[subsample]
 226        target_sub = target_train[subsample]
 227    else:
 228        features_sub = features_train
 229        target_sub = target_train
 230    baseline = _r_squared(
 231        target_sub,
 232        np.asarray(predict_fn(features_sub), dtype=float).reshape(-1),
 233    )
 234    drops = np.zeros(n_features, dtype=float)
 235    for feature_idx in range(n_features):
 236        shuffled = features_sub.copy()
 237        column_rng = np.random.default_rng(_IMPORTANCE_SEED + feature_idx)
 238        shuffled[:, feature_idx] = features_sub[
 239            column_rng.permutation(features_sub.shape[0]), feature_idx
 240        ]
 241        permuted = _r_squared(
 242            target_sub,
 243            np.asarray(predict_fn(shuffled), dtype=float).reshape(-1),
 244        )
 245        drops[feature_idx] = max(baseline - permuted, 0.0)
 246    total = float(drops.sum())
 247    if total > 0.0:
 248        return drops / total
 249    return np.full(n_features, 1.0 / n_features)

 255def purged_kfold_cv(
 256    features: Any,
 257    target: Any,
 258    n_splits: int = 5,
 259    purge_gap: int = 1,
 260    model_fn: ModelFn | None = None,
 261) -> dict[str, Any]:
 262    """Purged, contiguous K-fold CV for overlapping financial samples.
 263
 264    Folds are contiguous time blocks (no shuffling).  For each fold, training
 265    rows within ``purge_gap`` of either boundary of the test block are
 266    removed, so a training label whose horizon overlaps the test window
 267    cannot leak:
 268
 269        train indices kept:  j < test_lo - purge_gap  or
 270                              j > test_hi + purge_gap
 271
 272    Time-based variant with embargo: ``atropos._utils.purge_embargo_mask``.
 273
 274    When ``model_fn`` is given, every fold is scored: out-of-fold
 275    predictions, per-fold information coefficient (Pearson corr of forecast
 276    vs realized) and directional hit rate.  A fold whose training set is
 277    too small to fit (``n_train < n_features + 2``) is SKIPPED instead of
 278    silently scored as NaN: its fold row carries ``skipped_reason`` (a
 279    short string) while ``fold_ic`` / ``fold_hit`` stay NaN; scored folds
 280    carry ``skipped_reason = None``.
 281
 282    IMPORTANT — positional alignment: ``features`` and ``target`` are
 283    paired POSITIONALLY; pandas indexes are IGNORED (no reindexing, no
 284    sorting).  Rows must already be aligned and in CHRONOLOGICAL order —
 285    the contiguous fold blocks above are time blocks.
 286
 287    Returns ``{n, n_splits, purge_gap, n_features, folds (DataFrame),
 288    mean_ic, ic_t, ic_p_value, oos_hit_rate, oof_predictions,
 289    model_used}``.  ``ic_t`` is a t-stat of the per-fold ICs;
 290    ``ic_p_value`` is its two-sided p-value, ``2 * t.sf(|ic_t|,
 291    scored - 1)`` where ``scored`` is the number of folds actually
 292    scored (skipped folds must not inflate the degrees of freedom);
 293    ``oof_predictions`` is full-length with NaN outside
 294    test folds.
 295
 296    Example::
 297
 298        >>> rng = np.random.default_rng(0)  # doctest: +SKIP
 299        >>> feats = rng.standard_normal((500, 2))  # doctest: +SKIP
 300        >>> targ = (  # doctest: +SKIP
 301        ...     0.4 * feats[:, 0] + rng.standard_normal(500)
 302        ... )
 303        >>> cv = purged_kfold_cv(  # doctest: +SKIP
 304        ...     feats, targ, model_fn=ols_model_fn
 305        ... )
 306        >>> cv["mean_ic"] > 0.2  # doctest: +SKIP
 307        True
 308    """
 309    features_arr, target_arr, names = _as_arrays(features, target)
 310    n_rows, n_features = features_arr.shape
 311    ns = validate_int("n_splits", n_splits, minimum=2)
 312    gap = validate_int("purge_gap", purge_gap, minimum=0)
 313    if n_rows < 4 * ns:
 314        raise ValueError(
 315            f"n={n_rows} too small for {ns} folds (need >= 4*n_splits)"
 316        )
 317
 318    bounds = [
 319        np.asarray(part) for part in np.array_split(np.arange(n_rows), ns)
 320    ]
 321    oof = np.full(n_rows, np.nan) if model_fn is not None else None
 322    fold_rows: list[dict[str, Any]] = []
 323
 324    for fold_idx, te in enumerate(bounds):
 325        lo, hi = int(te[0]), int(te[-1])
 326        tr = np.concatenate(
 327            [bounds[other] for other in range(ns) if other != fold_idx]
 328        )
 329        tr = tr[(tr < lo - gap) | (tr > hi + gap)]
 330        row: dict[str, Any] = {
 331            "fold": fold_idx,
 332            "train_lo": int(tr.min()) if tr.size else -1,
 333            "train_hi": int(tr.max()) if tr.size else -1,
 334            "test_lo": lo,
 335            "test_hi": hi,
 336            "n_train": int(tr.size),
 337            "n_test": int(te.size),
 338        }
 339        min_train = n_features + 2
 340        if model_fn is not None and tr.size >= min_train:
 341            model = model_fn(features_arr[tr], target_arr[tr])
 342            preds = np.asarray(model(features_arr[te]), dtype=float).reshape(
 343                -1
 344            )
 345            if oof is not None:
 346                oof[te] = preds
 347            ic = _pearson(preds, target_arr[te])
 348            hit = float(np.mean(np.sign(preds) == np.sign(target_arr[te])))
 349            row["fold_ic"] = ic
 350            row["fold_hit"] = hit
 351            row["skipped_reason"] = None
 352        else:
 353            row["fold_ic"] = float("nan")
 354            row["fold_hit"] = float("nan")
 355            row["skipped_reason"] = (
 356                None
 357                if model_fn is None
 358                else f"n_train={int(tr.size)} < n_features+2={min_train}"
 359            )
 360        fold_rows.append(row)
 361
 362    folds = pd.DataFrame(fold_rows)
 363    ics = folds["fold_ic"].to_numpy(dtype=float)
 364    ics_valid = ics[np.isfinite(ics)]
 365    mean_ic = float(np.mean(ics_valid)) if ics_valid.size else float("nan")
 366    ic_t = float("nan")
 367    if ics_valid.size >= 2:
 368        se = float(np.std(ics_valid, ddof=1))
 369        ic_t = (
 370            float(mean_ic / se * np.sqrt(ics_valid.size))
 371            if se > 0
 372            else float("nan")
 373        )
 374    hits = folds["fold_hit"].to_numpy(dtype=float)
 375    hits_valid = hits[np.isfinite(hits)]
 376    return {
 377        "n": n_rows,
 378        "n_splits": ns,
 379        "purge_gap": gap,
 380        "n_features": n_features,
 381        "feature_names": names,
 382        "folds": folds,
 383        "mean_ic": mean_ic,
 384        "ic_t": ic_t,
 385        "ic_p_value": (
 386            float(2.0 * stats.t.sf(abs(ic_t), ics_valid.size - 1))
 387            if np.isfinite(ic_t)
 388            else float("nan")
 389        ),
 390        "oos_hit_rate": float(np.mean(hits_valid))
 391        if hits_valid.size
 392        else float("nan"),
 393        "oof_predictions": oof,
 394        "model_used": model_fn is not None,
 395    }

 401def walkforward_model_eval(
 402    features: Any,
 403    target: Any,
 404    model_fn: ModelFn,
 405    *,
 406    train_window: int = 180,
 407    test_window: int = 30,
 408    step: int | None = None,
 409    purge_gap: int = 0,
 410) -> dict[str, Any]:
 411    """Rolling train/test walk-forward: honest out-of-sample model quality.
 412
 413    IMPORTANT — positional alignment: ``features`` and ``target`` are
 414    paired POSITIONALLY; pandas indexes are IGNORED (no reindexing, no
 415    sorting).  Rows must already be aligned and in CHRONOLOGICAL order —
 416    the rolling windows below are time windows.
 417
 418    Every fold trains on the FIXED-length window
 419    ``[t0 - purge_gap - train_window, t0 - purge_gap)`` and tests on
 420    ``[t0, t0 + test_window)``; the window then slides by ``step``
 421    (default ``test_window``).  ``purge_gap`` rows directly before each
 422    test slice are removed from the train slice so overlapping
 423    forward-return labels cannot leak into the test period (set it to
 424    the label horizon; 1 for one-period forward returns).  Stacked
 425    out-of-sample forecasts drive:
 426
 427    * ``oos_ic``        - Pearson corr of forecasts vs realized outcomes,
 428    * ``pnl_t``         - t-stat of ``sign(forecast) * realized`` (long/short
 429      sign rule),
 430    * ``oos_hit_rate``  - share of folds-rows where sign(forecast) matches.
 431
 432    Verdict: PASS if ``pnl_t >= 2``; DEGRADED if ``>= 1``; FAIL otherwise.
 433
 434    Returns ``{n_folds, oos_ic, pnl_t, oos_hit_rate, verdict, folds
 435    (DataFrame), predictions, realized, train_window, test_window, step,
 436    purge_gap}``.
 437
 438    Example::
 439
 440        >>> rng = np.random.default_rng(0)  # doctest: +SKIP
 441        >>> feats = rng.standard_normal((500, 1))  # doctest: +SKIP
 442        >>> targ = (  # doctest: +SKIP
 443        ...     0.5 * feats[:, 0] + rng.standard_normal(500)
 444        ... )
 445        >>> wfe = walkforward_model_eval(  # doctest: +SKIP
 446        ...     feats, targ, ols_model_fn, train_window=150, test_window=50
 447        ... )
 448        >>> wfe["verdict"] in {"PASS", "DEGRADED", "FAIL"}  # doctest: +SKIP
 449        True
 450    """
 451    features_arr, target_arr, _ = _as_arrays(features, target)
 452    n_rows = features_arr.shape[0]
 453    tw = validate_int("train_window", train_window)
 454    ew = validate_int("test_window", test_window)
 455    st = validate_int("step", step) if step is not None else ew
 456    pg = validate_int("purge_gap", purge_gap, minimum=0)
 457    if tw < 10 or ew < 1 or st < 1:
 458        raise ValueError(
 459            "need train_window >= 10, test_window >= 1, step >= 1"
 460        )
 461    if tw + pg + ew > n_rows:
 462        raise ValueError(
 463            f"n={n_rows} too small for train_window={tw}"
 464            f" + purge_gap={pg} + test_window={ew}"
 465        )
 466
 467    preds_all: list[np.ndarray] = []
 468    real_all: list[np.ndarray] = []
 469    fold_rows: list[dict[str, Any]] = []
 470    t0 = tw + pg
 471    while t0 + ew <= n_rows:
 472        tr = slice(t0 - pg - tw, t0 - pg)
 473        te = slice(t0, t0 + ew)
 474        model = model_fn(features_arr[tr], target_arr[tr])
 475        preds = np.asarray(model(features_arr[te]), dtype=float).reshape(-1)
 476        realized = target_arr[te]
 477        if preds.size != realized.size:
 478            raise ValueError(
 479                f"predictor returned {preds.size} values for "
 480                f"{realized.size} rows"
 481            )
 482        pnl = np.sign(preds) * realized
 483        pnl_sd = float(np.std(pnl, ddof=1))
 484        fold_rows.append(
 485            {
 486                "fold": len(fold_rows),
 487                "train_lo": tr.start,
 488                "train_hi": tr.stop - 1,
 489                "test_lo": te.start,
 490                "test_hi": te.stop - 1,
 491                "n_train": tw,
 492                "n_test": ew,
 493                "purge_gap": pg,
 494                "fold_ic": _pearson(preds, realized),
 495                "fold_hit": float(
 496                    np.mean(np.sign(preds) == np.sign(realized))
 497                ),
 498                "fold_pnl_mean": float(np.mean(pnl)),
 499                "fold_pnl_t": (
 500                    float(np.mean(pnl) / pnl_sd * np.sqrt(pnl.size))
 501                    if pnl_sd > 0
 502                    else float("nan")
 503                ),
 504            }
 505        )
 506        preds_all.append(preds)
 507        real_all.append(realized)
 508        t0 += st
 509
 510    if not fold_rows:
 511        raise ValueError("no walk-forward folds produced")
 512    preds = np.concatenate(preds_all)
 513    realized = np.concatenate(real_all)
 514    pnl = np.sign(preds) * realized
 515    sd = float(np.std(pnl, ddof=1))
 516    pnl_t = (
 517        float(np.mean(pnl) / sd * np.sqrt(pnl.size))
 518        if sd > 0
 519        else float("nan")
 520    )
 521    hit = float(np.mean(np.sign(preds) == np.sign(realized)))
 522    if np.isfinite(pnl_t) and pnl_t >= 2.0:
 523        verdict = "PASS"
 524    elif np.isfinite(pnl_t) and pnl_t >= 1.0:
 525        verdict = "DEGRADED"
 526    else:
 527        verdict = "FAIL"
 528    return {
 529        "n_folds": len(fold_rows),
 530        "train_window": tw,
 531        "test_window": ew,
 532        "step": st,
 533        "purge_gap": pg,
 534        "oos_ic": _pearson(preds, realized),
 535        "pnl_t": pnl_t,
 536        "oos_hit_rate": hit,
 537        "verdict": verdict,
 538        "folds": pd.DataFrame(fold_rows),
 539        "predictions": preds,
 540        "realized": realized,
 541    }

 544# ----------------------------------------------------------------------------
 545# feature importance stability
 546# ----------------------------------------------------------------------------
 547def _extract_importances(
 548    model: Any,
 549    n_features: int,
 550    importance_fn: Callable[[Any], Any] | None,
 551) -> np.ndarray:
 552    """Pull a length-``n_features`` importance vector from a fitted model.
 553
 554    Dict models (entries) and object models (attributes) are probed with
 555    the SAME ordered key set ``("feature_importances_",
 556    "feature_importances", "coef_", "coef")`` and the SAME abs handling:
 557    coefficient-style keys (``coef_`` / ``coef``) are mapped through
 558    ``|.|`` because their sign carries no importance meaning;
 559    importance-style keys are returned as-is (downstream consumers such
 560    as :func:`feature_importance_stability` apply ``|.|`` themselves
 561    where needed).
 562
 563    Args:
 564        model: fitted model (dict or arbitrary object).
 565        n_features: expected length of the importance vector.
 566        importance_fn: explicit extractor; when given it fully replaces
 567            the default key probing.
 568
 569    Returns:
 570        ``(n_features,)`` importance vector.
 571
 572    Raises:
 573        ValueError: if no probed key yields a length-``n_features``
 574            vector and no ``importance_fn`` was supplied.
 575    """
 576    if importance_fn is not None:
 577        values = np.asarray(importance_fn(model), dtype=float).reshape(-1)
 578        if values.size != n_features:
 579            raise ValueError(
 580                f"importance_fn returned {values.size} values, "
 581                f"expected {n_features}"
 582            )
 583        return values
 584    for key in _IMPORTANCE_KEYS:
 585        if isinstance(model, dict):
 586            raw = model.get(key)
 587        else:
 588            raw = getattr(model, key, None)
 589        if raw is None:
 590            continue
 591        values = np.asarray(raw, dtype=float).reshape(-1)
 592        if values.size == n_features:
 593            if key in _COEFFICIENT_KEYS:
 594                # asarray pins the ndarray type (stubs -> Any).
 595                return np.asarray(np.abs(values))
 596            return values
 597    raise ValueError(
 598        "cannot extract feature importances from the fitted model; "
 599        "pass importance_fn(model) -> (n_features,) array"
 600    )

 603def feature_importance_stability(
 604    features: Any,
 605    target: Any,
 606    model_fn: ModelFn,
 607    *,
 608    n_bootstrap: int = 100,
 609    importance_fn: Callable[[Any], Any] | None = None,
 610    top_k: int = 1,
 611    seed: int | None = None,
 612) -> dict[str, Any]:
 613    """Are feature importances stable across bootstrap refits?
 614
 615    IMPORTANT — positional alignment: ``features`` and ``target`` are
 616    paired POSITIONALLY; pandas indexes are IGNORED (no reindexing, no
 617    sorting).  Rows must already be aligned and in CHRONOLOGICAL order
 618    (each bootstrap draw then resamples rows iid).
 619
 620    Each of ``n_bootstrap`` draws resamples rows (iid bootstrap), refits
 621    the model and extracts importances (default extractors:
 622    ``model.feature_importances_`` then ``|model.coef_|``; or an explicit
 623    ``importance_fn(model)``).  Vectors are normalized to sum 1 per draw,
 624    then summarized:
 625
 626    * per-feature mean / std / CV of importance, frequency in the top
 627      ``top_k`` ranks, mean descending rank;
 628    * ``mean_pairwise_spearman`` - mean rank correlation between every pair
 629      of bootstrap importance vectors.
 630
 631    Verdict: STABLE when at least 10 valid draws exist and the mean pairwise
 632    Spearman is >= 0.6; UNSTABLE otherwise (an UNSTABLE ranking means the
 633    "story" behind the model is not identifiable from this data).
 634
 635    Returns ``{per_feature (DataFrame), mean_pairwise_spearman, verdict,
 636    n_bootstrap, n_valid, n, n_features, top_k}``.
 637
 638    Example::
 639
 640        >>> rng = np.random.default_rng(0)  # doctest: +SKIP
 641        >>> feats = pd.DataFrame(  # doctest: +SKIP
 642        ...     {"a": rng.standard_normal(400), "b": rng.standard_normal(400)}
 643        ... )
 644        >>> targ = (  # doctest: +SKIP
 645        ...     0.6 * feats["a"] + rng.standard_normal(400)
 646        ... )
 647        >>> stab = feature_importance_stability(  # doctest: +SKIP
 648        ...     feats, targ, ols_model_fn, n_bootstrap=30, seed=0
 649        ... )
 650        >>> stab["verdict"]  # doctest: +SKIP
 651        'STABLE'
 652    """
 653    features_arr, target_arr, names = _as_arrays(features, target)
 654    n_rows, n_features = features_arr.shape
 655    nb = validate_int("n_bootstrap", n_bootstrap, minimum=2)
 656    tk = int(top_k)
 657    if not (1 <= tk <= n_features):
 658        raise ValueError(
 659            f"top_k must be in [1, n_features={n_features}], got {tk}"
 660        )
 661    tk = validate_int("top_k", top_k)
 662    rng = np.random.default_rng(seed)
 663
 664    imps: list[np.ndarray] = []
 665    for _ in range(nb):
 666        idx = rng.integers(0, n_rows, size=n_rows)
 667        model = model_fn(features_arr[idx], target_arr[idx])
 668        imp = _extract_importances(model, n_features, importance_fn)
 669        imp = np.abs(imp)
 670        total = float(imp.sum())
 671        if np.isfinite(imp).all() and total > 0:
 672            imps.append(imp / total)
 673    if len(imps) < 2:
 674        return {
 675            "per_feature": pd.DataFrame(),
 676            "mean_pairwise_spearman": float("nan"),
 677            "verdict": "UNSTABLE",
 678            "n_bootstrap": nb,
 679            "n_valid": len(imps),
 680            "n": n_rows,
 681            "n_features": n_features,
 682            "top_k": tk,
 683            "note": "fewer than 2 valid bootstrap draws",
 684        }
 685
 686    imps_arr = np.asarray(imps, dtype=float)
 687    mean = imps_arr.mean(axis=0)
 688    std = imps_arr.std(axis=0, ddof=1)
 689    with np.errstate(divide="ignore", invalid="ignore"):
 690        cv = np.where(np.abs(mean) > 1e-12, std / np.abs(mean), np.inf)
 691    desc_ranks = np.argsort(-imps_arr, axis=1)  # descending
 692    top_freq = np.array(
 693        [
 694            float(np.mean(desc_ranks[:, :tk] == feat_idx))
 695            for feat_idx in range(n_features)
 696        ]
 697    )
 698    mean_rank = np.array(
 699        [
 700            float(np.mean(np.where(desc_ranks == feat_idx)[1]) + 1.0)
 701            for feat_idx in range(n_features)
 702        ]
 703    )
 704
 705    ranks = np.apply_along_axis(
 706        stats.rankdata, 1, imps_arr
 707    )  # avg ranks, ascending
 708    rc = ranks - ranks.mean(axis=1, keepdims=True)
 709    cov = rc @ rc.T / float(n_features)
 710    sd = np.sqrt(np.diag(cov))
 711    with np.errstate(divide="ignore", invalid="ignore"):
 712        corr = np.where(np.outer(sd, sd) > 0, cov / np.outer(sd, sd), np.nan)
 713    iu = np.triu_indices(len(imps), k=1)
 714    rhos = corr[iu]
 715    rhos = rhos[np.isfinite(rhos)]
 716    mean_rho = float(np.mean(rhos)) if rhos.size else float("nan")
 717
 718    verdict = (
 719        "STABLE"
 720        if (len(imps) >= 10 and np.isfinite(mean_rho) and mean_rho >= 0.6)
 721        else "UNSTABLE"
 722    )
 723    per_feature = pd.DataFrame(
 724        {
 725            "feature": names,
 726            "mean_importance": mean,
 727            "std": std,
 728            "cv": cv,
 729            f"top{tk}_freq": top_freq,
 730            "mean_rank": mean_rank,
 731        }
 732    )
 733    return {
 734        "per_feature": per_feature,
 735        "mean_pairwise_spearman": mean_rho,
 736        "verdict": verdict,
 737        "n_bootstrap": nb,
 738        "n_valid": len(imps),
 739        "n": n_rows,
 740        "n_features": n_features,
 741        "top_k": tk,
 742    }

 745# ----------------------------------------------------------------------------
 746# sign consistency
 747# ----------------------------------------------------------------------------
 748def sign_consistency(
 749    predictions: Any,
 750    outcomes: Any,
 751    n_regimes: int = 3,
 752) -> dict[str, Any]:
 753    """Does the model get the SIGN right, everywhere (not just on average)?
 754
 755    ``directional_accuracy`` is the overall share of rows where
 756    ``sign(prediction) == sign(outcome)``.  ``sign_stability_by_regime``
 757    reports the same statistic within (a) ``n_regimes`` contiguous
 758    time blocks (keys ``regime_1`` ... ``regime_{n_regimes}``) and (b)
 759    terciles of ``|outcome|`` (calm / normal / violent rows, keys
 760    ``vol_tercile_low`` / ``vol_tercile_mid`` / ``vol_tercile_high``).
 761    A model that only works in one regime is a regime bet, not an edge.
 762
 763    IMPORTANT — positional alignment: ``predictions`` and ``outcomes``
 764    are paired POSITIONALLY; pandas indexes are IGNORED (no reindexing,
 765    no sorting).  Rows must already be aligned and in CHRONOLOGICAL
 766    order.
 767
 768    Degenerate inputs are rejected explicitly — a constant predictor is
 769    a beta bet, not a model, and a constant outcome has no sign
 770    variation to explain:
 771
 772    * constant predictions (``ptp(predictions) == 0``) -> verdict
 773      ``FAIL`` with ``reason="degenerate (constant) predictions"``;
 774    * constant outcomes (``ptp(outcomes) == 0``) -> verdict
 775      ``INCONCLUSIVE`` with ``reason="degenerate (constant) outcomes"``.
 776
 777    Exact zeros: ``sign(0) == 0`` carries no directional information, so
 778    rows where EITHER side is exactly 0 are EXCLUDED from every accuracy
 779    denominator (overall and per-regime); ``n_usable`` reports how many
 780    sign-bearing rows remain.  A regime with no usable rows is omitted
 781    from ``sign_stability_by_regime``.
 782
 783    Verdict: PASS when overall accuracy > 0.5 AND every regime accuracy
 784    > 0.5; FAIL otherwise (and always for degenerate predictions);
 785    INCONCLUSIVE for degenerate outcomes.
 786
 787    Returns ``{directional_accuracy, sign_stability_by_regime (dict),
 788    stable, verdict, n, n_usable}`` (plus ``reason`` in the degenerate
 789    cases and ``note`` when fewer than 10 finite rows remain).
 790
 791    Raises:
 792        ValueError: if ``n_regimes`` is not an integer >= 2 (no silent
 793            coercion; every other integer parameter is validated the
 794            same way).
 795
 796    Example::
 797
 798        >>> rng = np.random.default_rng(0)  # doctest: +SKIP
 799        >>> preds = rng.standard_normal(600)  # doctest: +SKIP
 800        >>> outcs = 0.5 * preds + rng.standard_normal(600)  # doctest: +SKIP
 801        >>> sign_consistency(preds, outcs)["verdict"]  # doctest: +SKIP
 802        'PASS'
 803    """
 804    pred_vals = np.asarray(predictions, dtype=float).reshape(-1)
 805    outc_vals = np.asarray(outcomes, dtype=float).reshape(-1)
 806    if pred_vals.size != outc_vals.size:
 807        raise ValueError(
 808            f"predictions has {pred_vals.size} values, "
 809            f"outcomes has {outc_vals.size}"
 810        )
 811    ok = np.isfinite(pred_vals) & np.isfinite(outc_vals)
 812    pred_vals, outc_vals = pred_vals[ok], outc_vals[ok]
 813    n_rows = int(pred_vals.size)
 814    if n_rows < 10:
 815        return {
 816            "directional_accuracy": float("nan"),
 817            "sign_stability_by_regime": {},
 818            "stable": False,
 819            "verdict": "FAIL",
 820            "n": n_rows,
 821            "n_usable": int(
 822                np.count_nonzero((pred_vals != 0.0) & (outc_vals != 0.0))
 823            ),
 824            "note": "insufficient data",
 825        }
 826    correct = np.sign(pred_vals) == np.sign(outc_vals)
 827    if float(np.ptp(pred_vals)) == 0.0:
 828        # "always +1" (or any constant) is a beta bet, not a model: it
 829        # must not PASS just because the outcome mean is positive.
 830        return {
 831            "directional_accuracy": float(np.mean(correct)),
 832            "sign_stability_by_regime": {},
 833            "stable": False,
 834            "verdict": "FAIL",
 835            "n": n_rows,
 836            "n_usable": int(np.count_nonzero(outc_vals != 0.0)),
 837            "reason": "degenerate (constant) predictions",
 838        }
 839    if float(np.ptp(outc_vals)) == 0.0:
 840        # constant outcome: sign accuracy is unidentifiable here
 841        return {
 842            "directional_accuracy": float(np.mean(correct)),
 843            "sign_stability_by_regime": {},
 844            "stable": False,
 845            "verdict": "INCONCLUSIVE",
 846            "n": n_rows,
 847            "n_usable": int(np.count_nonzero(pred_vals != 0.0)),
 848            "reason": "degenerate (constant) outcomes",
 849        }
 850    usable = (pred_vals != 0.0) & (outc_vals != 0.0)
 851    n_usable = int(np.count_nonzero(usable))
 852    acc = float(np.mean(correct[usable])) if n_usable else float("nan")
 853
 854    regime_acc: dict[str, float] = {}
 855    nr = validate_int("n_regimes", n_regimes, minimum=2)
 856    for regime_idx, chunk in enumerate(
 857        np.array_split(np.arange(n_rows), nr), start=1
 858    ):
 859        chunk_usable = usable[chunk]
 860        regime_acc[f"regime_{regime_idx}"] = (
 861            float(np.mean(correct[chunk][chunk_usable]))
 862            if chunk_usable.any()
 863            else float("nan")
 864        )
 865    mag_bins = np.clip(
 866        np.digitize(
 867            np.abs(outc_vals),
 868            np.quantile(np.abs(outc_vals), [1.0 / 3.0, 2.0 / 3.0]),
 869        ),
 870        0,
 871        2,
 872    )
 873    for name, bin_idx in zip(
 874        ("vol_tercile_low", "vol_tercile_mid", "vol_tercile_high"),
 875        (0, 1, 2),
 876        strict=True,  # 3 regime names for 3 tercile bins, by construction
 877    ):
 878        sel = (mag_bins == bin_idx) & usable
 879        if sel.any():
 880            regime_acc[name] = float(np.mean(correct[sel]))
 881
 882    stable = (
 883        bool(all(entry > 0.5 for entry in regime_acc.values()))
 884        and len(regime_acc) > 0
 885    )
 886    verdict = "PASS" if (acc > 0.5 and stable) else "FAIL"
 887    return {
 888        "directional_accuracy": acc,
 889        "sign_stability_by_regime": regime_acc,
 890        "stable": stable,
 891        "verdict": verdict,
 892        "n": n_rows,
 893        "n_usable": n_usable,
 894    }

 897# ----------------------------------------------------------------------------
 898# deflated performance
 899# ----------------------------------------------------------------------------
 900def deflated_performance(
 901    returns: Any,
 902    n_strategies: int,
 903    mdd: float,
 904    periods: int = 252,
 905    sr_var_annual: Any = 1.0,
 906) -> dict[str, Any]:
 907    """Sharpe deflated for multiple testing, with drawdown context.
 908
 909    Delegates the statistics to ``evaluation_metrics.deflated_sharpe``
 910    (Bailey & Lopez de Prado, 2014): the observed Sharpe is compared against
 911    the expected MAXIMUM Sharpe of ``n_strategies`` unit-variance trials, so
 912    a "best of N" selection cannot masquerade as an edge.  ``n_strategies
 913    = 1`` disables the multiple-testing penalty (DSR == PSR): with a single
 914    trial there is no "best of N" selection to punish.  ``mdd`` (the
 915    strategy's max drawdown, negative) turns the excess Sharpe into a
 916    drawdown-scaled ``deflated_calmar`` diagnostic.
 917
 918    Verdict: PASS when the deflated-Sharpe p-value < 0.05 (the edge survives
 919    having tried ``n_strategies`` strategies); FAIL otherwise.
 920
 921    Args:
 922        returns: per-period return series of the selected strategy.
 923        n_strategies: number of trials the strategy was selected from;
 924            ``1`` disables the multiple-testing penalty (DSR == PSR).
 925        mdd: max drawdown of the strategy, NEGATIVE by convention
 926            (``mdd == 0`` means "no drawdown observed" and yields
 927            ``deflated_calmar = inf`` for a positive excess Sharpe).
 928        periods: periods per year (252 for daily data).
 929        sr_var_annual: variance of Sharpe ratios across trials, in
 930            annualized units (forwarded to ``deflated_sharpe``).
 931
 932    Returns:
 933        ``{dsr, p_value, sr_annual, sr_benchmark_annual, excess_sr, mdd,
 934        deflated_calmar, n_strategies, verdict}``.
 935
 936    Raises:
 937        ValueError: if ``n_strategies`` or ``periods`` is not an
 938            integer >= 1, or ``mdd > 0`` (a positive drawdown is a
 939            sign error, not a drawdown).
 940
 941    Example::
 942
 943        >>> rng = np.random.default_rng(0)  # doctest: +SKIP
 944        >>> rets = rng.normal(0.0008, 0.01, 1500)  # doctest: +SKIP
 945        >>> dp = deflated_performance(  # doctest: +SKIP
 946        ...     rets, n_strategies=10, mdd=-0.15
 947        ... )
 948        >>> dp["verdict"]  # doctest: +SKIP
 949        'PASS'
 950    """
 951    ns = validate_int("n_strategies", n_strategies)
 952    periods = validate_int("periods", periods)
 953    mdd_f = validate_number("mdd", mdd)
 954    if mdd_f > 0.0:
 955        raise ValueError(
 956            f"mdd must be <= 0 (max drawdown is negative), got {mdd_f}"
 957        )
 958    base = deflated_sharpe(
 959        returns, n_trials=ns, periods=periods, sr_var_annual=sr_var_annual
 960    )
 961    sr_a = float(base["sr_annual"])
 962    sr0 = float(base["sr_benchmark_annual"])
 963    excess = (
 964        sr_a - sr0 if np.isfinite(sr_a) and np.isfinite(sr0) else float("nan")
 965    )
 966    if np.isfinite(excess) and np.isfinite(mdd_f) and mdd_f < 0:
 967        deflated_calmar = float(excess / abs(mdd_f))
 968    elif np.isfinite(excess) and excess > 0:
 969        deflated_calmar = float("inf")
 970    else:
 971        deflated_calmar = float("nan")
 972    p_value = float(base["p_value"])
 973    verdict = "PASS" if (np.isfinite(p_value) and p_value < 0.05) else "FAIL"
 974    return {
 975        "dsr": float(base["dsr"]),
 976        "p_value": p_value,
 977        "sr_annual": sr_a,
 978        "sr_benchmark_annual": sr0,
 979        "excess_sr": excess,
 980        "mdd": mdd_f,
 981        "deflated_calmar": deflated_calmar,
 982        "n_strategies": ns,
 983        "verdict": verdict,
 984    }

 987# ----------------------------------------------------------------------------
 988# default model factory
 989# ----------------------------------------------------------------------------
 990def ols_model_fn(
 991    features_train: np.ndarray, target_train: np.ndarray
 992) -> Predictor:
 993    """Default ``model_fn``: OLS on ``[1 | features]``, returns a predictor.
 994
 995    The predictor carries ``coef_`` (fitted coefficients incl. intercept)
 996    and ``feature_importances_`` (|beta_j| * sd(feature_j), normalized to
 997    sum 1) so the default importance extractor of
 998    :func:`feature_importance_stability` works without configuration.
 999
1000    The fit is fail-closed on rank deficiency: with collinear,
1001    duplicated or constant feature columns the least-squares solution
1002    is not unique, so the fit raises instead of silently returning a
1003    minimum-norm coefficient pick.
1004
1005    Raises:
1006        ValueError: if the design matrix ``[1 | features]`` is
1007            rank-deficient (collinear features) - the coefficients are
1008            not identifiable.
1009
1010    Example::
1011
1012        >>> model = ols_model_fn(  # doctest: +SKIP
1013        ...     features_train, target_train
1014        ... )
1015        >>> preds = model(features_test)  # doctest: +SKIP
1016    """
1017    features_arr = np.asarray(features_train, dtype=float)
1018    if features_arr.ndim == 1:
1019        features_arr = features_arr.reshape(-1, 1)
1020    target_arr = np.asarray(target_train, dtype=float).reshape(-1)
1021    design = np.column_stack([np.ones(target_arr.size), features_arr])
1022    if np.linalg.matrix_rank(design) < design.shape[1]:
1023        raise ValueError(
1024            "design is rank-deficient (collinear, duplicated or "
1025            "constant feature columns): OLS coefficients are not "
1026            "identifiable - drop or de-duplicate the offending columns"
1027        )
1028    beta, _, _, _ = np.linalg.lstsq(design, target_arr, rcond=None)
1029    sd = features_arr.std(axis=0)
1030    sd = np.where(sd > 0, sd, 1.0)
1031    imp = np.abs(beta[1:] * sd)
1032    total = float(imp.sum())
1033    imp_norm = imp / total if total > 0 else imp
1034
1035    def predict(features_test: Any, _coeffs: np.ndarray = beta) -> np.ndarray:
1036        test_arr = np.asarray(features_test, dtype=float)
1037        if test_arr.ndim == 1:
1038            test_arr = test_arr.reshape(-1, 1)
1039        forecasts = (
1040            np.column_stack([np.ones(test_arr.shape[0]), test_arr]) @ _coeffs
1041        )
1042        return np.asarray(forecasts, dtype=float)
1043
1044    predict.coef_ = beta  # type: ignore[attr-defined]
1045    predict.feature_importances_ = imp_norm  # type: ignore[attr-defined]
1046    return predict

1049# ----------------------------------------------------------------------------
1050# native non-parametric model factories (numpy only)
1051# ----------------------------------------------------------------------------
1052def knn_model_fn(
1053    features_train: np.ndarray,
1054    target_train: np.ndarray,
1055    n_neighbors: int = 5,
1056) -> Predictor:
1057    """Distance-weighted k-NN regressor packaged as a ``model_fn``.
1058
1059    Features are standardized with the train mean/std (zero std -> 1.0)
1060    inside the fit.  A forecast is the inverse-distance weighted mean
1061    of the ``n_neighbors`` nearest training targets::
1062
1063        prediction = sum(w_i * y_i) / sum(w_i),
1064        w_i = 1 / (d_i + 1e-12)
1065
1066    with euclidean distances ``d_i`` measured in standardized space.
1067
1068    ``feature_importances_`` is permutation importance on a capped
1069    train subsample (<= 256 rows drawn without replacement when the
1070    train set is larger, fixed seed 0): the drop in R^2 after shuffling
1071    one feature column (per-column generator seeded ``0 + column``),
1072    clipped at 0 and normalized to sum 1 (uniform when every drop is
1073    0).  Fitting is fully deterministic.
1074
1075    Args:
1076        features_train: ``(n_train, n_features)`` feature matrix.
1077        target_train: ``(n_train,)`` outcome vector.
1078        n_neighbors: number of nearest training rows averaged per
1079            forecast.
1080
1081    Returns:
1082        Predictor closure with attributes ``n_neighbors_`` (validated
1083        neighbor count) and ``feature_importances_``.
1084
1085    Raises:
1086        ValueError: if ``n_neighbors < 1`` or it exceeds the number of
1087            training rows.
1088
1089    Example::
1090
1091        >>> model = knn_model_fn(  # doctest: +SKIP
1092        ...     features_train, target_train, n_neighbors=5
1093        ... )
1094        >>> preds = model(features_test)  # doctest: +SKIP
1095    """
1096    features_arr = _as_matrix(features_train)
1097    target_arr = np.asarray(target_train, dtype=float).reshape(-1)
1098    n_rows = features_arr.shape[0]
1099    neighbors = validate_int("n_neighbors", n_neighbors)
1100    if n_rows < 1:
1101        raise ValueError("need at least 1 training row")
1102    if neighbors > n_rows:
1103        raise ValueError(
1104            f"n_neighbors={neighbors} exceeds n_train={n_rows} rows"
1105        )
1106    train_std, center, spread = _standardize_fit(features_arr)
1107
1108    def predict(features_test: Any) -> np.ndarray:
1109        test_std = (_as_matrix(features_test) - center) / spread
1110        dist_sq = _pairwise_dist_sq(test_std, train_std)
1111        neighbor_idx = np.argpartition(dist_sq, neighbors - 1, axis=1)[
1112            :, :neighbors
1113        ]
1114        neighbor_dist = np.sqrt(
1115            np.take_along_axis(dist_sq, neighbor_idx, axis=1)
1116        )
1117        weights = 1.0 / (neighbor_dist + 1e-12)
1118        weighted = weights * target_arr[neighbor_idx]
1119        return np.asarray(
1120            weighted.sum(axis=1) / weights.sum(axis=1), dtype=float
1121        )
1122
1123    importances = _permutation_importance(predict, features_arr, target_arr)
1124    predict.n_neighbors_ = neighbors  # type: ignore[attr-defined]
1125    predict.feature_importances_ = importances  # type: ignore[attr-defined]
1126    return predict

1129def kernel_model_fn(
1130    features_train: np.ndarray,
1131    target_train: np.ndarray,
1132    bandwidth: float | None = None,
1133) -> Predictor:
1134    """Nadaraya-Watson gaussian-kernel regressor as a ``model_fn``.
1135
1136    Features are standardized with the train mean/std (zero std -> 1.0)
1137    inside the fit.  A forecast is the kernel-weighted mean over ALL
1138    training rows::
1139
1140        prediction = sum(K_i * y_i) / sum(K_i),
1141        K_i = exp(-0.5 * (d_i / bandwidth)**2)
1142
1143    with euclidean distances ``d_i`` in standardized space; when the
1144    kernel weight sum is numerically negligible (``denom <= 1e-12``, e.g.
1145    every ``K_i`` underflows to 0 for a query very far from all training
1146    rows) the forecast falls back to the single nearest training target.
1147
1148    The default bandwidth is Scott's rule ``n ** (-1 / (n_features +
1149    4))`` computed on the standardized train data (scalar).  Importances
1150    use the same capped deterministic permutation scheme as
1151    :func:`knn_model_fn`.
1152
1153    Args:
1154        features_train: ``(n_train, n_features)`` feature matrix.
1155        target_train: ``(n_train,)`` outcome vector.
1156        bandwidth: positive kernel width in standardized units;
1157            ``None`` selects Scott's rule.
1158
1159    Returns:
1160        Predictor closure with attributes ``bandwidth_`` (effective
1161        width) and ``feature_importances_``.
1162
1163    Raises:
1164        ValueError: if an explicit ``bandwidth`` is not > 0.
1165
1166    Example::
1167
1168        >>> model = kernel_model_fn(  # doctest: +SKIP
1169        ...     features_train, target_train, bandwidth=0.5
1170        ... )
1171        >>> preds = model(features_test)  # doctest: +SKIP
1172    """
1173    features_arr = _as_matrix(features_train)
1174    target_arr = np.asarray(target_train, dtype=float).reshape(-1)
1175    n_rows, n_features = features_arr.shape
1176    if n_rows < 1:
1177        raise ValueError("need at least 1 training row")
1178    train_std, center, spread = _standardize_fit(features_arr)
1179    if bandwidth is None:
1180        width = float(n_rows ** (-1.0 / (n_features + 4)))
1181    else:
1182        width = validate_number("bandwidth", bandwidth)
1183        if not width > 0.0:
1184            raise ValueError(f"bandwidth must be > 0, got {width}")
1185
1186    def predict(features_test: Any) -> np.ndarray:
1187        test_std = (_as_matrix(features_test) - center) / spread
1188        dist_sq = _pairwise_dist_sq(test_std, train_std)
1189        weights = np.exp(-0.5 * dist_sq / (width * width))
1190        denom = weights.sum(axis=1)
1191        # denom in (0, 1e-12] is numerically useless (the weights are
1192        # pure underflow noise): fall back to the nearest training row.
1193        usable_denom = denom > 1e-12
1194        smoothed = (weights @ target_arr) / np.where(usable_denom, denom, 1.0)
1195        nearest = target_arr[np.argmin(dist_sq, axis=1)]
1196        return np.asarray(
1197            np.where(usable_denom, smoothed, nearest), dtype=float
1198        )
1199
1200    importances = _permutation_importance(predict, features_arr, target_arr)
1201    predict.bandwidth_ = width  # type: ignore[attr-defined]
1202    predict.feature_importances_ = importances  # type: ignore[attr-defined]
1203    return predict

def feature_importance_stability( features: Any, target: Any, model_fn: Callable[[numpy.ndarray, numpy.ndarray], Callable[[typing.Any], numpy.ndarray]], *, n_bootstrap: int = 100, importance_fn: Callable[[typing.Any], typing.Any] | None = None, top_k: int = 1, seed: int | None = None) -> dict[str, typing.Any]: 
604def feature_importance_stability(
605    features: Any,
606    target: Any,
607    model_fn: ModelFn,
608    *,
609    n_bootstrap: int = 100,
610    importance_fn: Callable[[Any], Any] | None = None,
611    top_k: int = 1,
612    seed: int | None = None,
613) -> dict[str, Any]:
614    """Are feature importances stable across bootstrap refits?
615
616    IMPORTANT — positional alignment: ``features`` and ``target`` are
617    paired POSITIONALLY; pandas indexes are IGNORED (no reindexing, no
618    sorting).  Rows must already be aligned and in CHRONOLOGICAL order
619    (each bootstrap draw then resamples rows iid).
620
621    Each of ``n_bootstrap`` draws resamples rows (iid bootstrap), refits
622    the model and extracts importances (default extractors:
623    ``model.feature_importances_`` then ``|model.coef_|``; or an explicit
624    ``importance_fn(model)``).  Vectors are normalized to sum 1 per draw,
625    then summarized:
626
627    * per-feature mean / std / CV of importance, frequency in the top
628      ``top_k`` ranks, mean descending rank;
629    * ``mean_pairwise_spearman`` - mean rank correlation between every pair
630      of bootstrap importance vectors.
631
632    Verdict: STABLE when at least 10 valid draws exist and the mean pairwise
633    Spearman is >= 0.6; UNSTABLE otherwise (an UNSTABLE ranking means the
634    "story" behind the model is not identifiable from this data).
635
636    Returns ``{per_feature (DataFrame), mean_pairwise_spearman, verdict,
637    n_bootstrap, n_valid, n, n_features, top_k}``.
638
639    Example::
640
641        >>> rng = np.random.default_rng(0)  # doctest: +SKIP
642        >>> feats = pd.DataFrame(  # doctest: +SKIP
643        ...     {"a": rng.standard_normal(400), "b": rng.standard_normal(400)}
644        ... )
645        >>> targ = (  # doctest: +SKIP
646        ...     0.6 * feats["a"] + rng.standard_normal(400)
647        ... )
648        >>> stab = feature_importance_stability(  # doctest: +SKIP
649        ...     feats, targ, ols_model_fn, n_bootstrap=30, seed=0
650        ... )
651        >>> stab["verdict"]  # doctest: +SKIP
652        'STABLE'
653    """
654    features_arr, target_arr, names = _as_arrays(features, target)
655    n_rows, n_features = features_arr.shape
656    nb = validate_int("n_bootstrap", n_bootstrap, minimum=2)
657    tk = int(top_k)
658    if not (1 <= tk <= n_features):
659        raise ValueError(
660            f"top_k must be in [1, n_features={n_features}], got {tk}"
661        )
662    tk = validate_int("top_k", top_k)
663    rng = np.random.default_rng(seed)
664
665    imps: list[np.ndarray] = []
666    for _ in range(nb):
667        idx = rng.integers(0, n_rows, size=n_rows)
668        model = model_fn(features_arr[idx], target_arr[idx])
669        imp = _extract_importances(model, n_features, importance_fn)
670        imp = np.abs(imp)
671        total = float(imp.sum())
672        if np.isfinite(imp).all() and total > 0:
673            imps.append(imp / total)
674    if len(imps) < 2:
675        return {
676            "per_feature": pd.DataFrame(),
677            "mean_pairwise_spearman": float("nan"),
678            "verdict": "UNSTABLE",
679            "n_bootstrap": nb,
680            "n_valid": len(imps),
681            "n": n_rows,
682            "n_features": n_features,
683            "top_k": tk,
684            "note": "fewer than 2 valid bootstrap draws",
685        }
686
687    imps_arr = np.asarray(imps, dtype=float)
688    mean = imps_arr.mean(axis=0)
689    std = imps_arr.std(axis=0, ddof=1)
690    with np.errstate(divide="ignore", invalid="ignore"):
691        cv = np.where(np.abs(mean) > 1e-12, std / np.abs(mean), np.inf)
692    desc_ranks = np.argsort(-imps_arr, axis=1)  # descending
693    top_freq = np.array(
694        [
695            float(np.mean(desc_ranks[:, :tk] == feat_idx))
696            for feat_idx in range(n_features)
697        ]
698    )
699    mean_rank = np.array(
700        [
701            float(np.mean(np.where(desc_ranks == feat_idx)[1]) + 1.0)
702            for feat_idx in range(n_features)
703        ]
704    )
705
706    ranks = np.apply_along_axis(
707        stats.rankdata, 1, imps_arr
708    )  # avg ranks, ascending
709    rc = ranks - ranks.mean(axis=1, keepdims=True)
710    cov = rc @ rc.T / float(n_features)
711    sd = np.sqrt(np.diag(cov))
712    with np.errstate(divide="ignore", invalid="ignore"):
713        corr = np.where(np.outer(sd, sd) > 0, cov / np.outer(sd, sd), np.nan)
714    iu = np.triu_indices(len(imps), k=1)
715    rhos = corr[iu]
716    rhos = rhos[np.isfinite(rhos)]
717    mean_rho = float(np.mean(rhos)) if rhos.size else float("nan")
718
719    verdict = (
720        "STABLE"
721        if (len(imps) >= 10 and np.isfinite(mean_rho) and mean_rho >= 0.6)
722        else "UNSTABLE"
723    )
724    per_feature = pd.DataFrame(
725        {
726            "feature": names,
727            "mean_importance": mean,
728            "std": std,
729            "cv": cv,
730            f"top{tk}_freq": top_freq,
731            "mean_rank": mean_rank,
732        }
733    )

826        }

890        "sign_stability_by_regime": regime_acc,
891        "stable": stable,

893        "n": n_rows,
894        "n_usable": n_usable,
